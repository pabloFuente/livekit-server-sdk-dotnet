# '*' is a wildcard. Always add a reason, so the next /sync-protocol run knows why.

livekit.CloudAgent/*            # livekit_cloud_agent.proto, managed by the lk CLI; only server-sdk-go wraps it, with an AgentGrant this SDK does not have
